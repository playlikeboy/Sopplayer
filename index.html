import { Sopplayer } from 'sopplayer';
import 'sopplayer/css';

const player = new Sopplayer('#video-element', {
  theme: 'flamingo',
  keyboard: true
});
import { useEffect, useRef } from 'react';
import { Sopplayer } from 'sopplayer';
import 'sopplayer/css';

export function VideoPlayer({ src, poster }) {
  const videoRef = useRef(null);
  const playerRef = useRef(null);

  useEffect(() => {
    playerRef.current = new Sopplayer(videoRef.current, {
      theme: 'default',
      aspectRatio: '16:9'
    });
    return () => playerRef.current?.destroy();
  }, [src]);

  return <video ref={videoRef} poster={poster}>
    <source src={src} type="video/mp4" />
  </video>;
}

Why it feels light

Small on the page. Powerful under the hood.

Useful features without turning a simple video player into a framework of its own.

🪶

Zero runtime dependencies

Pure JavaScript at runtime. Your users download a small player instead of a dependency tree.

♿

Accessibility first

Keyboard navigation, focus states, slider semantics, live announcements, captions and reduced motion support.

⌨️

Keyboard friendly

Play, pause, seek, volume, fullscreen, PiP, captions and playback speed without leaving the keyboard.

🎨

Seven built-in themes

Switch visual identity instantly or use CSS custom properties to make your own skin.

🔒

Hardened by design

Media URLs and user-controlled values are handled defensively to reduce common DOM injection risks.

🔁

Friendly to older apps

Legacy integration paths and the Video.js-style compatibility layer make migration less painful.
